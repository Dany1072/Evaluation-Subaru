<!-- Script de gestion des étapes et de l'envoi EmailJS -->
<script>
let currentStep = 1;
let vehicleDataSummary = "";

function updateStepIndicator() {
document.getElementById('step-indicator').innerText = `Étape ${currentStep} sur 6`;
}

async function nextStep(step) {
if (step === 1) {
const vin = document.getElementById('vin-input').value.trim();
if (vin.length < 5) {
alert("Veuillez entrer un numéro de série valide.");
return;
}
// Décodage VIN en temps réel via API NHTSA
try {
const response = await fetch(`https://vpic.nhtsa.dot.gov/api/vehicles/decodevinvalues/${vin}?format=json`);
const data = await response.json();
if (data.Results && data.Results.length > 0) {
const r = data.Results[0];
const year = r.ModelYear || '';
const make = r.Make || '';
const model = r.Model || '';
const trim = r.Trim || '';
vehicleDataSummary = `${year} ${make} ${model} ${trim}`.trim();
if (vehicleDataSummary.length > 2) {
document.getElementById('vehicle-specs').innerText = vehicleDataSummary;
document.getElementById('vehicle-preview').classList.remove('hidden');
} else {
vehicleDataSummary = "Véhicule standard (NIV: " + vin + ")";
document.getElementById('vehicle-specs').innerText = vehicleDataSummary;
document.getElementById('vehicle-preview').classList.remove('hidden');
}
}
} catch(e) {
vehicleDataSummary = "NIV: " + vin;
}
}

document.getElementById(`step-${currentStep}`).classList.add('hidden');
currentStep++;
document.getElementById(`step-${currentStep}`).classList.remove('hidden');
updateStepIndicator();
}

function prevStep(step) {
document.getElementById(`step-${currentStep}`).classList.add('hidden');
currentStep--;
document.getElementById(`step-${currentStep}`).classList.remove('hidden');
updateStepIndicator();
}

// Soumission finale du formulaire via EmailJS
document.getElementById('eval-form').addEventListener('submit', function(e) {
e.preventDefault();

const btn = document.getElementById('submit-btn');
btn.innerText = "Envoi en cours...";
btn.disabled = true;

const templateParams = {
to_email: "dany.leclerc@levissubaru.com",
client_name: document.getElementById('client-name').value,
client_contact: document.getElementById('client-contact').value,
vin: document.getElementById('vin-input').value,
vehicule_info: vehicleDataSummary,
kilometrage: document.getElementById('kilometrage').value,
q_keys: document.getElementById('q-keys').value,
q_accident: document.getElementById('q-accident').value,
q_windshield: document.getElementById('q-windshield').value,
q_wheels: document.getElementById('q-wheels').value,
comments: document.getElementById('comments').value || "Aucun commentaire",
photos_links: "Photos transmises dans le dossier client (vérification visuelle en concession)."
};

// Envoi via vos identifiants EmailJS
emailjs.send("service_8cnhcxf", "template_ef327qf", templateParams)
.then(function(response) {
document.getElementById(`step-${currentStep}`).classList.add('hidden');
currentStep = 6;
document.getElementById('step-6').classList.remove('hidden');
document.getElementById('step-indicator').innerText = "Terminé";
}, function(error) {
alert("Une erreur est survenue lors de l'envoi. Veuillez réessayer.");
btn.innerText = "Soumettre";
btn.disabled = false;
});
});
</script>
